class Solution {
public:
    bool isAnagram(string s, string t) {
        int record[26] = {0};
        for(int i = 0; i<s.size();i++){
            record[s[i] - 'a']++;
        }
        for(int i = 0; i<t.size(); i++){
            record[t[i]-'a']--;
        }
        for(int i = 0; i<26; i++){
            if(record[i]!=0) return false;
        }
        return true;
    }
};

//字符-‘a'为ASC码
//记录数组不为零说明有字母不一样
